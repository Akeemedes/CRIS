cmake_minimum_required(VERSION 3.20)
project(BratuTraining LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Torch REQUIRED)
find_package(OpenMP REQUIRED)
if(NOT TARGET MKL::MKL)
  find_package(MKL CONFIG REQUIRED)
endif()
add_executable(bratu_lm_trainer src/bratu_lm_trainer.cpp)
target_include_directories(bratu_lm_trainer PRIVATE ../native_lm/include)
target_link_libraries(bratu_lm_trainer PRIVATE ${TORCH_LIBRARIES} MKL::MKL OpenMP::OpenMP_CXX)
target_compile_options(bratu_lm_trainer PRIVATE ${TORCH_CXX_FLAGS})
if(MSVC)
  target_compile_options(bratu_lm_trainer PRIVATE /O2 /EHsc /W4)
endif()
