cmake_minimum_required(VERSION 3.22)
project(FrozenLinearSolver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(frozen_linear_solver frozen_linear_solver.cpp)
target_include_directories(frozen_linear_solver PRIVATE ../../src)
target_compile_definitions(frozen_linear_solver PRIVATE AMGCL_NO_BOOST)
if(MSVC)
  target_compile_options(frozen_linear_solver PRIVATE /O2 /EHsc)
endif()
